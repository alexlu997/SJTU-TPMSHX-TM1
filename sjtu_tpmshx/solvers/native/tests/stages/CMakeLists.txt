cmake_minimum_required(VERSION 3.31.8)
project(tpmshx_stage_consumer LANGUAGES CXX)
enable_testing()
find_package(tpmshx CONFIG REQUIRED COMPONENTS preparation solver result)
foreach(smoke IN ITEMS stage_run_smoke quick_design_stage_smoke)
    add_executable(${smoke} ../${smoke}.cpp)
    target_compile_features(${smoke} PRIVATE cxx_std_17)
    target_link_libraries(${smoke} PRIVATE tpmshx::preparation tpmshx::solver tpmshx::result)
    if(WIN32)
        foreach(component IN ITEMS preparation solver result)
            add_custom_command(TARGET ${smoke} POST_BUILD
                COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                    "$<TARGET_FILE:tpmshx::${component}>" "$<TARGET_FILE_DIR:${smoke}>")
        endforeach()
    endif()
    if(smoke STREQUAL "stage_run_smoke")
        set(result_directory "${CMAKE_BINARY_DIR}/stage-result")
    else()
        set(result_directory "${CMAKE_BINARY_DIR}/quick-design-stage-result")
    endif()
    add_test(NAME ${smoke} COMMAND ${smoke} "${result_directory}")
endforeach()
